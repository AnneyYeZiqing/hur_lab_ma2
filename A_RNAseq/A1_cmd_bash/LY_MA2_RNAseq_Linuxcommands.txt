#IMPORTANT: before trimming, copy TruSeq3-PE-2.fa to the current directory
#adapter trimming
### trimmomatic.sh:
#!/bin/bash
#SBATCH -p short
#SBATCH -c 4
#SBATCH -t 1:30:00
#SBATCH --mem=16G
#SBATCH -e trim.%j.err
#SBATCH -o trim.%j.out

module load conda/miniforge3
module load trimmomatic/0.39

ADAPTER_FA="./TruSeq3-PE-2.fa"

if [[ ! -f "$ADAPTER_FA" ]]; then
    echo "ERROR: Trimmomatic adapter file not found: $ADAPTER_FA" >&2
    exit 1
fi

OUTDIR="../A2_trimmed_paired"
mkdir -p "$OUTDIR"

sample_path="$1"
dir=$(dirname "$sample_path")
sample=$(basename "$sample_path")

java -jar "$TRIMMOMATIC/trimmomatic-0.39.jar" PE -threads 4 -phred33 "${dir}/${sample}_R1.fastq.gz" "${dir}/${sample}_R2.fastq.gz" "${OUTDIR}/output_${sample}_R1_paired.fq.gz" "${OUTDIR}/output_${sample}_R1_unpaired.fq.gz" "${OUTDIR}/output_${sample}_R2_paired.fq.gz" "${OUTDIR}/output_${sample}_R2_unpaired.fq.gz" ILLUMINACLIP:"$ADAPTER_FA":2:30:10 LEADING:3 TRAILING:3 MINLEN:36


base=()
for fq in */*_R1.fastq.gz; do
    base+=("${fq%_R1.fastq.gz}")
done
for fq in "${base[@]}"; do echo "$fq"; done
for fq in "${base[@]}"; do
    sbatch trimmomatic.sh "$fq"
done

mkdir A2_trimmed_paired/unpaired
mv A2_trimmed_paired/output*_unpaired.fq.gz A2_trimmed_paired/unpaired/

### countsplit.sh:
#!/bin/bash
input_file="$1"
awk -F'\t' 'BEGIN {OFS = "\t"}
NR == 1 {
    next
}
NR == 2 {
    for (col = 7; col <= NF; col++) {
        # Strip quotes from the header string
        header = $col
        gsub(/"/, "", header)
        split(header, path_parts, "/")
        stripped_header = path_parts[3]
        split(stripped_header, name_parts, "_")
        sample_name = name_parts[2]

        # replacing "output" with the sample name
        out_filename = "output_gene_counts.txt"
        gsub(/output/, sample_name, out_filename)
        output_file[col] = out_filename

        print "Geneid", "Chr", "Start", "End", "Strand", "Length", "Count" > output_file[col]
    }
    next
}
{
    for (col = 7; col <= NF; col++) {
        print $1, $2, $3, $4, $5, $6, $col > output_file[col]
    }
}
' "$input_file"
